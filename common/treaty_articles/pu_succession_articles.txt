# Personal Union Rework - succession treaty articles


pu_recognize_succession = {
	kind = directed
	icon = "gfx/interface/icons/law_icons/monarchy.dds"

	cost = 0
	maintenance_paid_by = target_country
	usage_limit = once_per_side

	max_target_involvement = 2500

	flags = {
		can_be_enforced
	}

	possible = {
		pu_recognize_succession_possible = {
			REQUESTER = scope:other_country
		}
	}

	on_entry_into_force = {
		scope:article_options = {
			target_country = {
				ruler = {
					save_scope_as = pu_source_ruler
				}
			}

			source_country = {
				if = {
					limit = {
						exists = heir
					}

					heir = {
						add_character_role = pu_displaced_heir
					}
				}

				set_heir = scope:pu_source_ruler
			}
		}
	}

	ai = {
		treaty_categories = {
			expansion
		}

		article_ai_usage = {
			request
		}

		evaluation_chance = {
			value = 0.65

			add = {
				value = country_rank
				multiply = 0.05
			}
		}

		# ============================================================
		# STRUCTURAL VALUE
		# ============================================================

		inherent_accept_score = {
			value = 0

			# We are doing this to be able to reuse loc across treaties and diplomatic actions
			scope:source_country = {
				save_temporary_scope_as = target
			}
			scope:target_country = {
				save_temporary_scope_as = actor
			}

			# --------------------------------------------------------
			# SOURCE COUNTRY
			# Country being asked to recognize the foreign ruler
			# as its heir.
			# --------------------------------------------------------
			if = {
				limit = {
					scope:source_country ?= this
				}

				# This should normally be a major concession.
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -300
				}

				# Shared primary culture makes a dynastic arrangement
				if = {
					limit = {
						any_primary_culture = {
							is_primary_culture_of = scope:target_country
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_SAME_CULTURE"
						value = 50
					}
				}
				else_if = {
					### If you share a cultural discrimination trait
					limit = {
						any_primary_culture = {
							shares_heritage_trait_with_any_primary_culture = scope:target_country
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_CULTURAL_ACCEPTANCE"
						value = 30
					}
				}
				else_if = {
					### If you share a cultural discrimination trait
					limit = {
						any_primary_culture = {
							shares_heritage_trait_group_with_any_primary_culture = scope:target_country
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_CULTURAL_ACCEPTANCE"
						value = 20
					}
				}
				else_if = {
					### If you share a cultural discrimination trait
					limit = {
						any_primary_culture = {
							scope:target_country = {
								"cultural_acceptance_base(prev)" >= 40
							}
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DIFFERENT_CULTURE"
						value = 10
					}
				}

				# Shared state religion.
				if = {
					limit = {
						religion = scope:target_country.religion
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_SAME_RELIGION"
						value = 40
					}
				}
				else_if = {
					### If you share a religious discrimination trait
					limit = {
						religion = {
							shares_heritage_trait_with_state_religion = scope:target_country
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DIFFERENT_RELIGION"
						value = 20
					}
				}

				add = {
					value = country_turmoil
					multiply = 200
					max = 100
					desc = "COUNTRY_TURMOIL"
				}

				# ====================================================
				# Financial condition
				#
				# Financial distress reduces resistance, but normally
				# does not make Recognize Succession desirable by itself.
				# ====================================================

				if = {
					limit = {
						in_default = yes
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DEBT_TARGET"
						value = 120
					}
				}
				else = {
					# Debt: up to +100.
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DEBT_TARGET"

						value = scaled_debt
						multiply = 200

						max = 100
					}

					# Large gold reserves make submission less attractive:
					# up to -50.
					subtract = {
						desc = "DIPLOMATIC_ACCEPTANCE_GOLD_RESERVES"

						value = scaled_gold_reserves
						multiply = 200

						max = 50
					}
				}

				if = {
					limit = {
						has_modifier = declared_bankruptcy
					}

					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DEBT_TARGET"
						value = 50
					}
				}

				# GDP comparison

				if = {
					limit = {
						gdp >= 2000000
					}

					subtract = {
						desc = "DIPLOMATIC_ACCEPTANCE_HIGH_GDP"
						value = gdp
						subtract = 1000000
						divide = 1000000
						min = 1
						max = 50
					}
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_GDP"
					value = "scope:target_country.ai_gdp_comparison(scope:source_country)"
					multiply = 10
					min = -40
					max = 40
				}

				# Army power comparison

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ARMY_STRENGTH"
					value = "scope:target_country.ai_army_comparison(scope:source_country)"
					multiply = 10
					min = -50
					max = 50
				}


				# ====================================================
				# Attitude toward requester
				# ====================================================

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:target_country
									attitude = antagonistic
								}
								has_attitude = {
									who = scope:target_country
									attitude = domineering
								}
								has_attitude = {
									who = scope:target_country
									attitude = belligerent
								}
								has_attitude = {
									who = scope:target_country
									attitude = rebellious
								}
							}
						}
						value = -1000
					}
					else_if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:target_country
									attitude = wary
								}
								has_attitude = {
									who = scope:target_country
									attitude = disinterested
								}
								has_attitude = {
									who = scope:target_country
									attitude = cautious
								}
							}
						}
						value = -100
					}
				}
			}

			# --------------------------------------------------------
			# TARGET COUNTRY
			# Country requesting recognition of its ruler.
			#
			# Positive acceptance represents how much value the AI
			# is willing to give up elsewhere in the treaty.
			# --------------------------------------------------------
			else = {
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -25
				}

				# Value of gaining a future Personal Union over source.
				add = {
					desc = "PU_ACCEPTANCE_THRONE_VALUE"
					value = scope:source_country.country_infamy_value
					multiply = 1.5
					max = 100
				}

				# ====================================================
				# Attitude toward the country whose succession it wants
				# ====================================================

				# Domineering attitude strongly values political control.

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:source_country
									attitude = domineering
								}
								has_attitude = {
									who = scope:source_country
									attitude = protective
								}
								has_attitude = {
									who = scope:source_country
									attitude = cooperative
								}
								has_attitude = {
									who = scope:source_country
									attitude = genial
								}
							}
						}

						add = 50
					}
				}
			}
		}

		wargoal_score_multiplier = {
			value = 0.2

			if = {
				### If you share a cultural discrimination trait
				limit = {
					any_primary_culture = {
						shares_heritage_trait_group_with_any_primary_culture = scope:target_country
					}
				}
				add = 0.4
			}

			if = {
				### If you share a religious discrimination trait
				limit = {
					religion = {
						shares_heritage_trait_with_state_religion = scope:target_country
					}
				}
				add = 0.4
			}

			if = {
				limit = {
					scope:target_country = {
						OR = {
							play_participant_has_war_goal_of_type_against = {
								type = make_protectorate
								target = scope:source_country
							}
							play_participant_has_war_goal_of_type_against = {
								type = make_tributary
								target = scope:source_country
							}
							play_participant_has_war_goal_of_type_against = {
								type = make_personal_union
								target = scope:source_country
							}
						}
					}
				}
				multiply = 0
			}
		}

		proposal_weight = {
			value = 1
			if = {
				limit = {
					scope:other_country = {
						OR = {
							ai_is_regional_objective_subjugate_target = { TARGET = root }
							ai_is_regional_objective_protect_target = { TARGET = root }
						}
					}
				}
				add = 5
			}
		}
	}

	wargoal = {
		execution_priority = 55
		contestion_type = control_half_target_country_states

		maneuvers = {
			add = {
				desc = "MANEUVERS_BASE_VALUE"
				value = 10
			}

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
					max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		infamy_severity = 0.4

		infamy = {
			value = pu_recognize_succession_infamy
		}
	}

}


pu_renounce_succession = {
	kind = directed
	icon = "gfx/interface/icons/war_goals/independence.dds"

	cost = 0
	maintenance_paid_by = source_country
	usage_limit = once_per_side

	flags = {
		can_be_enforced
		hostile
	}

	required_inputs = {
		country
	}

	possible = {
		exists = ruler
		highest_overlapping_interest_tier = { target = scope:other_country value >= interest_tier_effect_can_start_diplomatic_plays_min_tier }
		country_has_monarchy_law = yes
	}

	country_valid_trigger = {
		# input = country whose throne the source ruler is heir to
		scope:input = {
			exists = heir
			heir ?= root.ruler
		}
	}

	on_entry_into_force = {
		scope:article_options = {
			input_country = {
				if = {
					limit = {
						any_character = {
							has_role = pu_displaced_heir
						}
					}

					random_character = {
						limit = {
							has_role = pu_displaced_heir
						}

						save_scope_as = pu_restored_heir
					}

					set_heir = scope:pu_restored_heir

					scope:pu_restored_heir = {
						remove_character_role = pu_displaced_heir
					}
				}
				else = {
					create_character = {
						heir = yes
					}
				}
			}
		}
	}

	ai = {
		treaty_categories = {
			other
		}

		article_ai_usage = {
			request
		}

		evaluation_chance = {
			value = 0.1
		}

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					scope:source_country ?= this
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -50
				}

				# Political value of the throne being renounced.
				subtract = {
					desc = "PU_ACCEPTANCE_THRONE_VALUE"
					value = scope:article.input_country.country_infamy_value
					multiply = 1.5
					max = 300
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"

					if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:target_country
									attitude = antagonistic
								}

								has_attitude = {
									who = scope:target_country
									attitude = domineering
								}

								has_attitude = {
									who = scope:target_country
									attitude = belligerent
								}

								has_attitude = {
									who = scope:target_country
									attitude = rebellious
								}
							}
						}

						value = -1000
					}
				}
			}
			else = {
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -10
				}

				if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:article.input_country
								attitude = domineering
							}
							has_attitude = {
								who = scope:article.input_country
								attitude = protective
							}
						}
					}

					add = {
						desc = "PU_ACCEPTANCE_THRONE_VALUE"
						value = scope:article.input_country.country_infamy_value
						multiply = 1.5
						max = 30
					}
				}
			}
		}

		wargoal_score_multiplier = {
			value = 1
		}
	}

	wargoal = {
		execution_priority = 56
		contestion_type = control_any_target_country_state

		maneuvers = {
			value = 10

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
					max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		infamy_severity = 0.10

		infamy = {
			value = 0

			scope:target_country = {
				add = country_infamy_value
			}
		}
	}
}


pu_force_abdication = {
	kind = directed
	icon = "gfx/interface/icons/war_goals/regime_change.dds"

	cost = 0
	maintenance_paid_by = source_country
	usage_limit = once_per_side

	flags = {
		can_be_enforced
		hostile
	}

	possible = {
		exists = ruler
		exists = heir
		highest_overlapping_interest_tier = { target = scope:other_country value >= interest_tier_effect_can_start_diplomatic_plays_min_tier }

		scope:other_country = {
			exists = ruler
			root.heir ?= ruler
		}
	}

	can_ratify = {
		scope:source_country = {
			exists = heir
			heir ?= scope:target_country.ruler
		}
	}

	on_entry_into_force = {
		scope:article_options = {
			source_country = {
				ruler = {
					retire_character = yes
				}
			}
		}
	}

	ai = {
		treaty_categories = {
			expansion
		}

		article_ai_usage = {
			request
		}

		evaluation_chance = {
			value = 0.1
		}

		inherent_accept_score = {
			value = 0

			if = {
				limit = {
					scope:source_country ?= this
				}

				# Even a recognized heir normally has to wait for succession.
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = -200
				}

				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"

					if = {
						limit = {
							OR = {
								has_attitude = {
									who = scope:target_country
									attitude = antagonistic
								}

								has_attitude = {
									who = scope:target_country
									attitude = domineering
								}

								has_attitude = {
									who = scope:target_country
									attitude = belligerent
								}

								has_attitude = {
									who = scope:target_country
									attitude = rebellious
								}
							}
						}

						value = -1000
					}
				}
			}
			else = {
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_BASE_ADDITIONAL"
					value = 25
				}
				add = {
					desc = "PU_ACCEPTANCE_THRONE_VALUE"
					value = scope:source_country.country_infamy_value
					multiply = 1.5
					max = 200
				}
			}
		}

		wargoal_score_multiplier = {
			value = 1
		}
	}

	wargoal = {
		execution_priority = 57
		contestion_type = control_half_target_country_states

		maneuvers = {
			value = 10

			scope:target_country = {
				multiply = {
					desc = "MANEUVERS_TARGET_COUNTRY_POPULATION_FACTOR"
					value = total_population
					divide = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_FACTOR
					multiply = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER
					max = define:NDiplomacy|SWAY_OFFER_WARGOAL_MANEUVERS_COST_POPULATION_SCALING_MULTIPLIER_MAX_TOTAL
				}
			}
		}

		# Succession has already been legally recognized.
		# This only accelerates it, so infamy is much lower.
		infamy_severity = 0.20

		infamy = {
			value = pu_recognize_succession_infamy
		}
	}
}
