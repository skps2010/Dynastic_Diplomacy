on_become_subject = {
	on_actions = {
		pu_personal_union_established
	}
}

pu_personal_union_established = {
	trigger = {
		is_subject_type = subject_type_personal_union
	}

	effect = {
		# root = new junior partner
		save_scope_as = pu_junior

		scope:overlord = {
			save_scope_as = pu_senior
		}

		# Notify every human player
		every_country = {
			limit = {
				is_player = yes
			}

			post_notification = personal_union_established_message
		}
	}
}

on_new_ruler = {
	on_actions = {
		pu_update_recognized_succession
	}
}

pu_update_recognized_succession = {
	effect = {
		save_temporary_scope_as = pu_new_ruler

		owner = {
			save_temporary_scope_as = pu_new_ruler_country

			if = {
				limit = {
					country_has_monarchy_law = yes
				}

				every_scope_treaty = {
					every_scope_article = {
						limit = {
							has_type = pu_recognize_succession

							target_country = {
								this ?= scope:pu_new_ruler_country
							}
						}

						source_country = {
							if = {
								limit = {
									country_has_monarchy_law = yes
								}

								set_heir = scope:pu_new_ruler
							}
						}
					}
				}
			}
		}
	}
}
